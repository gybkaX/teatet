string perevod(int b, int n) {
    // n это число в 10 b система в которую нужно перевести

    string result = "";

    for (int i = 0; n > 0; i++) {
        if (10 == b) {
            result = result;
            break;
        }

        else {
            if (n % b > 9) {
                result = char('A' + (n % b) - 10) + result;
            }
            else {
                result = char((n % b) + '0') + result;
            }
        }

        n /= b;
    }

    return result;
}
